const fetchData = async (stateId?, stateName?) => {
    const items = [];
    let lastPage = false
    for (let start = 0; start < totalItems && !lastPage; start += pageSize) {
        let payload;
        if (stateId) {
            payload = {
                start,
                length: pageSize,
                searchTag:"",
                filters:{
                    state_id:[stateId],
                    authority_id:[]}}
        } else {
            payload = {
                start,
                length: pageSize,
                sortBy: "name",
                filters: {
                ministry_id: [],
                department_id: [],
                },
            };
        }

        try {
            const response = await axios.post(url, payload);
            const data = response.data;
            if (data.length < pageSize) {
                lastPage = true; // If the number of items returned is less than pageSize, it's the last page
            }

            for (const item of data || []) {
                let rawTitle;
                if (stateId) {
                    rawTitle = item["stateUrlTitle"] || "";
                } else {
                    rawTitle = item["urlTitle"] || "";
                }
                const cleanTitle = rawTitle.replace(/-/g, " ");
                allTitles.push(cleanTitle);
                items.push(item); // Add full item to items array
            }
        } catch (error) {
            console.error(`Error fetching data from ${start} to ${start + pageSize}:`, error);
        }
    }

    // Write items to a JSON file
    const parser = new Parser(); // This will use all keys from the first object as columns
